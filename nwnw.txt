Candidates (Other) = DISTINCTCOUNT('Candidates Other'[Name])
Attended (Other) = CALCULATE([Candidates (Other)], 'Candidates Other'[Technical] IN {"Passed", "Failed"})
Passed (Other) = CALCULATE([Candidates (Other)], 'Candidates Other'[Outcome] = "Accepted")
Pass Rate (Other) = DIVIDE([Passed (Other)], [Attended (Other)])
No Shows (Other) = CALCULATE([Candidates (Other)], 'Candidates Other'[Outcome] = "No show")
Offers Made (Other) = COUNTROWS('Offers Other')
Offers Accepted (Other) = CALCULATE([Offers Made (Other)], 'Offers Other'[Offer Group] = "Accepted")
Offers In Progress (Other) = CALCULATE([Offers Made (Other)], 'Offers Other'[Offer Group] = "In progress")
Offers Lost (Other) = CALCULATE([Offers Made (Other)], 'Offers Other'[Offer Group] = "Lost")
Joined (Other) = CALCULATE([Offers Made (Other)], 'Offers Other'[Offer Status] = "Joined")
