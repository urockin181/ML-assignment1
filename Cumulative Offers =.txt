Cumulative Offers =
VAR CurrentWave =
    VALUE ( SUBSTITUTE ( MAX ( Waves[Wave] ), "Wave ", "" ) )
VAR CurrentGroups =
    VALUES ( Offers[Offer Group] )
RETURN
    CALCULATE (
        COUNTROWS ( Offers ),
        REMOVEFILTERS ( Waves ),
        REMOVEFILTERS ( Offers ),
        CurrentGroups,
        FILTER (
            ALL ( Offers[Wave] ),
            IFERROR ( VALUE ( TRIM ( SUBSTITUTE ( UPPER ( Offers[Wave] ), "WAVE", "" ) ) ), 999 ) <= CurrentWave
        )
    )
