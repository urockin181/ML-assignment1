"""
HAL Weekly Report generator.

Recreates the weekly HAL PowerPoint report from:
  1. A CSV export of call records (Date, Queue, Total Queue columns).
  2. An Excel export of tickets (Primary Configuration Item column).
  3. A No CI Found Excel export, used only if that week broke the streak.
  4. A C-SAT Excel export (any .xlsx with "C-SAT" in its name, found
     automatically in the CSV's folder), read from its Summary sheet.

Usage:
    python generate_report.py [--csv PATH] [--excel PATH] [--output PATH] [--noci PATH]

If --csv/--excel/--noci are omitted, the DEFAULT_* paths below are used
(edit them to match the actual filenames each week). If --output is
omitted, the file is named
"HAL Weekly Report - <Start Date> to <End Date>.pptx" and saved inside
the DEFAULT_OUTPUT_PATH folder, using the dates you enter below.

The script then interactively asks for the same six values the original
script used: Start Day Date, End Day Date, Start Date Month, End Date Month,
Start Date Year, End Date Year, plus a No CI Found This Week (yes/no)
prompt. Answering "yes" increments a streak counter persisted in
STREAK_STATE_PATH and shows it in a sentence; answering "no" resets the
streak and lists the actual records from the No CI Found file instead.

Slide design (colors, fonts, layout) is copied from Sample_design.pptx:
navy/gold color scheme, "Quattrocento Sans" font, an eyebrow section
label + bold title + accent line header, and a footer with the report
date range + page number. Charts are native PowerPoint shapes (bars
drawn as rectangles), not images.
"""

def parse_month(value):
    value = value.strip()
    if value.isdigit():
        month = int(value)
        if 1 <= month <= 12:
            return month
        raise ValueError(f"Month number out of range: {value}")
    for fmt_names in (calendar.month_name, calendar.month_abbr):
        for i, name in enumerate(fmt_names):
            if name and name.lower() == value.lower():
                return i
    raise ValueError(f"Could not parse month: {value}")


def prompt_int(label):
    while True:
        raw = input(f"{label}: ").strip()
        if raw.isdigit():
            return int(raw)
        print("Please enter a number.")


def prompt_month(label):
    while True:
        raw = input(f"{label}: ").strip()
        try:
            return parse_month(raw)
        except ValueError as exc:
            print(exc)


def prompt_yes_no(label):
    while True:
        raw = input(f"{label} (yes/no): ").strip().lower()
        if raw in ("y", "yes"):
            return True
        if raw in ("n", "no"):
            return False
        print("Please enter yes or no.")

def resolve_csv_path(path):
    """If `path` is a folder, returns the .csv file inside it (the most
    recently modified one, if there's more than one). Returns `path`
    unchanged if it's already a file."""
    path = Path(path)
    if not path.is_dir():
        return path
    csv_files = sorted(path.glob("*.csv"), key=lambda p: p.stat().st_mtime, reverse=True)
    if not csv_files:
        raise FileNotFoundError(f"No .csv file found in folder: {path}")
    if len(csv_files) > 1:
        names = ", ".join(p.name for p in csv_files)
        print(f"Multiple CSV files found in {path} ({names}); using the most recently modified: {csv_files[0].name}")
    return csv_files[0]


def resolve_csat_path(folder):
    """Finds the .xlsx file whose name contains "C-SAT" (any spacing/case)
    inside `folder`. Returns None if no such file is found."""
    folder = Path(folder)

    def norm(s):
        return re.sub(r"[^a-z0-9]", "", s.lower())

    matches = [p for p in folder.glob("*.xlsx") if "csat" in norm(p.stem)]
    if not matches:
        return None
    if len(matches) > 1:
        matches = sorted(matches, key=lambda p: p.stat().st_mtime, reverse=True)
        names = ", ".join(p.name for p in matches)
        print(f"Multiple C-SAT files found in {folder} ({names}); using the most recently modified: {matches[0].name}")
    return matches[0]


def load_calls(csv_path, start_date, end_date):
    df = pd.read_csv(csv_path)

    df["ParsedDate"] = pd.to_datetime(df["Date"], format="%m/%d/%y %I:%M %p", errors="coerce")
    df = df.dropna(subset=["ParsedDate"])

    # Only count calls that landed in one of the HAL queues.
    df = df[df["Queue"].isin(VALID_QUEUES)].copy()

    end_of_day = end_date + pd.Timedelta(hours=23, minutes=59, seconds=59)
    df = df[(df["ParsedDate"] >= start_date) & (df["ParsedDate"] <= end_of_day)]

    df["Weekday"] = pd.Categorical(df["ParsedDate"].dt.day_name(), categories=WEEKDAYS, ordered=True)
    df["Hour"] = df["ParsedDate"].dt.hour

    total_queue = df["Total Queue"].astype(str).str.strip()
    df["TotalQueueSeconds"] = pd.to_timedelta(total_queue, errors="coerce").dt.total_seconds()

    return df


def load_apps(xlsx_path):
    df = pd.read_excel(xlsx_path)
    items = df["Primary Configuration Item"].dropna().astype(str)
    items = items.apply(lambda s: re.sub(r"[​‌‍﻿]", "", s).strip())
    items = items[items != ""]
    return items.value_counts().head(TOP_N_APPS)


def _clean_text(series):
    return series.fillna("").astype(str).apply(lambda s: re.sub(r"[​‌‍﻿]", "", s).strip())


def load_pivot_table(xlsx_path):
    """Primary Configuration Item / Summary / Number of records, for the
    fixed PIVOT_ITEMS whitelist, sorted by item then by count descending."""
    df = pd.read_excel(xlsx_path)
    df = df.assign(
        **{"Primary Configuration Item": _clean_text(df["Primary Configuration Item"])}
    )
    df = df[df["Primary Configuration Item"].isin(PIVOT_ITEMS)].copy()
    df["Summary"] = _clean_text(df["Summary"])
    df = df[df["Summary"] != ""]

    pivot = (
        df.groupby(["Primary Configuration Item", "Summary"])
        .size()
        .reset_index(name="Number")
    )
    pivot["Primary Configuration Item"] = pd.Categorical(
        pivot["Primary Configuration Item"], categories=PIVOT_ITEMS, ordered=True
    )
    pivot = pivot.sort_values(
        ["Primary Configuration Item", "Number"], ascending=[True, False]
    ).reset_index(drop=True)
    return pivot


def load_noci_records(xlsx_path):
    """Raw "No CI Found" ticket records, with text columns cleaned and the
    Opened column formatted as a readable date string."""
    df = pd.read_excel(xlsx_path)
    for col in df.columns:
        if df[col].dtype == object:
            df[col] = _clean_text(df[col])
    if "Opened" in df.columns:
        opened = pd.to_datetime(df["Opened"], errors="coerce")
        df["Opened"] = opened.dt.strftime("%d %b %Y %H:%M").fillna(df["Opened"].astype(str))
    return df


# --------------------------------------------------------------------------- #
# C-SAT chart extraction (reads the named Excel chart objects' own cell
# references, so it stays correct even if the underlying data rows shift)
# --------------------------------------------------------------------------- #

_NS_R = "http://schemas.openxmlformats.org/officeDocument/2006/relationships"
_NS_MAIN = "http://schemas.openxmlformats.org/spreadsheetml/2006/main"
_NS_XDR = "http://schemas.openxmlformats.org/drawingml/2006/spreadsheetDrawing"
_NS_CHART_REL = "http://schemas.openxmlformats.org/drawingml/2006/chart"


def _get_named_charts(xlsx_path, sheet_name):
    """Returns {chart_name: openpyxl_chart_object} for every chart on
    `sheet_name`, keyed by Excel's own internal chart object name (e.g.
    "Chart 5"), by cross-referencing the sheet's drawing XML."""
    wb = openpyxl.load_workbook(xlsx_path, data_only=True)
    ws = wb[sheet_name]
    charts = ws._charts
    if not charts:
        return {}

    with zipfile.ZipFile(xlsx_path) as z:
        wb_xml = ET.fromstring(z.read("xl/workbook.xml"))
        sheet_rid = None
        for sheet_el in wb_xml.find(f"{{{_NS_MAIN}}}sheets"):
            if sheet_el.get("name") == sheet_name:
                sheet_rid = sheet_el.get(f"{{{_NS_R}}}id")
                break
        if sheet_rid is None:
            return {}

        wb_rels = ET.fromstring(z.read("xl/_rels/workbook.xml.rels"))
        sheet_target = next((rel.get("Target") for rel in wb_rels if rel.get("Id") == sheet_rid), None)
        if sheet_target is None:
            return {}
        sheet_target = sheet_target.lstrip("/")
        if not sheet_target.startswith("xl/"):
            sheet_target = "xl/" + sheet_target

        sheet_dir, sheet_file = sheet_target.rsplit("/", 1)
        sheet_rels_path = f"{sheet_dir}/_rels/{sheet_file}.rels"
        if sheet_rels_path not in z.namelist():
            return {}
        sheet_rels = ET.fromstring(z.read(sheet_rels_path))
        drawing_target = next((rel.get("Target") for rel in sheet_rels if "drawing" in rel.get("Type", "")), None)
        if drawing_target is None:
            return {}
        drawing_target = drawing_target.replace("../", "xl/")
        if not drawing_target.startswith("xl/"):
            drawing_target = "xl/drawings/" + drawing_target.rsplit("/", 1)[-1]

        drawing_xml = ET.fromstring(z.read(drawing_target))
        name_rid_pairs = []
        for gf in drawing_xml.iter(f"{{{_NS_XDR}}}graphicFrame"):
            cnvpr = gf.find(f".//{{{_NS_XDR}}}nvGraphicFramePr/{{{_NS_XDR}}}cNvPr")
            chart_el = gf.find(f".//{{{_NS_CHART_REL}}}chart")
            if cnvpr is not None and chart_el is not None:
                name_rid_pairs.append((cnvpr.get("name"), chart_el.get(f"{{{_NS_R}}}id")))

        drawing_dir, drawing_file = drawing_target.rsplit("/", 1)
        drawing_rels_path = f"{drawing_dir}/_rels/{drawing_file}.rels"
        drawing_rels = ET.fromstring(z.read(drawing_rels_path))
        rid_to_target = {rel.get("Id"): rel.get("Target") for rel in drawing_rels}

        ordered_names = [name for name, _ in name_rid_pairs]

    if len(ordered_names) != len(charts):
        return {}
    return dict(zip(ordered_names, charts))


def _resolve_ref_values(wb, ref):
    """Resolves a chart cell reference like "Summary!$B$8" or
    "Summary!$B$8:$F$8" into a flat list of live cell values."""
    sheet_part, cell_part = ref.split("!")
    sheet_part = sheet_part.strip("'")
    ws = wb[sheet_part]
    cell_part = cell_part.replace("$", "")
    cells = ws[cell_part]
    if isinstance(cells, tuple):
        flat = []
        for row in cells:
            if isinstance(row, tuple):
                flat.extend(c.value for c in row)
            else:
                flat.append(row.value)
        return flat
    return [cells.value]


def extract_named_chart_data(xlsx_path, sheet_name, chart_name):
    """Returns (categories, series) for the Excel chart object named
    `chart_name` on `sheet_name`, where series is a list of
    (series_name, [values]) using the chart's own live cell references."""
    wb = openpyxl.load_workbook(xlsx_path, data_only=True)
    named_charts = _get_named_charts(xlsx_path, sheet_name)
    chart = named_charts.get(chart_name)
    if chart is None:
        raise ValueError(f"Chart named '{chart_name}' not found on sheet '{sheet_name}'")

    categories = None
    series_data = []
    for s in chart.series:
        series_name = None
        if s.tx is not None and s.tx.strRef is not None:
            names = _resolve_ref_values(wb, s.tx.strRef.f)
            series_name = str(names[0]) if names else None
        values = [v if v is not None else 0 for v in _resolve_ref_values(wb, s.val.numRef.f)]
        if categories is None and s.cat is not None:
            if s.cat.strRef is not None:
                categories = [str(v) for v in _resolve_ref_values(wb, s.cat.strRef.f)]
            elif s.cat.numRef is not None:
                categories = [str(v) for v in _resolve_ref_values(wb, s.cat.numRef.f)]
        series_data.append((series_name, values))

    if categories is None:
        categories = [str(i) for i in range(len(series_data[0][1]))] if series_data else []
    return categories, series_data

def seconds_to_mmss(seconds):
    if pd.isna(seconds):
        seconds = 0
    minutes, sec = divmod(int(round(seconds)), 60)
    return f"{minutes:02d}:{sec:02d}"


def hour_label(hour):
    return pd.Timestamp(year=2000, month=1, day=1, hour=hour).strftime("%I %p").lstrip("0")


def ordinal(n):
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"

def format_date_range(start_date, end_date):
    if start_date.date() == end_date.date():
        return format_date(start_date)
    return f"{format_date(start_date)} - {format_date(end_date)}"


def cover_date_range(start_date, end_date):
    """Short form for the cover slide's big headline, e.g. "11 – 17 Aug 2026"
    (same month) or "27 Jul – 2 Aug 2026" (crossing months)."""
    if start_date.month == end_date.month and start_date.year == end_date.year:
        return f"{start_date.day} – {end_date.day} {calendar.month_abbr[end_date.month]} {end_date.year}"
    return (f"{start_date.day} {calendar.month_abbr[start_date.month]} – "
            f"{end_date.day} {calendar.month_abbr[end_date.month]} {end_date.year}")


def clean_vote_label(label):
    """"Strongly Disagree(1)" -> "Strongly disagree" (strip the trailing
    "(N)" and switch to sentence case, matching the reference design)."""
    label = re.sub(r"\s*\(\d+\)\s*$", "", label).strip()
    if not label:
        return label
    return label[0].upper() + label[1:].lower()


_VOTE_COLOR_BY_LABEL = {
    "strongly disagree": RED,
    "disagree": LIGHT_RED,
    "neutral": ORANGE,
    "agree": LIGHT_GREEN,
    "strongly agree": GREEN,
}


def vote_colors(labels):
    """One color per (already-cleaned) vote label, matched by sentiment
    text when recognized, falling back to the Strongly-Disagree-to-
    Strongly-Agree color order by position otherwise."""
    return [
        _VOTE_COLOR_BY_LABEL.get(label.lower(), VOTES_SENTIMENT_COLORS[i % len(VOTES_SENTIMENT_COLORS)])
        for i, label in enumerate(labels)
    ]


def date_for_weekday(start_date, end_date, day):
    """First calendar date within [start_date, end_date] whose weekday name is `day`."""
    d = start_date
    while d <= end_date:
        if d.day_name() == day:
            return d
        d += pd.Timedelta(days=1)
    return None


def _hour_range(df):
    if df.empty:
        return list(range(24))
    return list(range(int(df["Hour"].min()), int(df["Hour"].max()) + 1))

def add_rect(slide, left, top, width, height, color):
    """A plain solid-fill rectangle with no outline."""
    shape = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, left, top, max(int(width), 1), max(int(height), 1))
    shape.fill.solid()
    shape.fill.fore_color.rgb = color
    shape.line.fill.background()
    shape.shadow.inherit = False
    return shape


def add_chart_border(slide, left, top, width, height, color=LIGHT_GRAY):
    """A light, unfilled rectangle framing a chart's plot area."""
    shape = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, left, top, max(int(width), 1), max(int(height), 1))
    shape.fill.background()
    shape.line.color.rgb = color
    shape.line.width = Pt(1.5)
    shape.shadow.inherit = False
    return shape


def add_text(slide, left, top, width, height, text, size, color, bold=False,
             align=PP_ALIGN.LEFT, anchor=MSO_ANCHOR.TOP, word_wrap=False, rotation=0):
    box = slide.shapes.add_textbox(left, top, width, height)
    tf = box.text_frame
    tf.word_wrap = word_wrap
    tf.vertical_anchor = anchor
    tf.margin_left = 0
    tf.margin_right = 0
    tf.margin_top = 0
    tf.margin_bottom = 0
    tf.text = text
    p = tf.paragraphs[0]
    p.alignment = align
    if p.runs:
        run = p.runs[0]
        run.font.name = FONT_NAME
        run.font.size = Pt(size)
        run.font.bold = bold
        run.font.color.rgb = color
    if rotation:
        box.rotation = rotation
    return box

def new_slide(prs, blank_layout, eyebrow, title, footer_text, page_num, logo_path):
    slide = prs.slides.add_slide(blank_layout)

    if logo_path is not None:
        slide.shapes.add_picture(str(logo_path), HEADER_LOGO_LEFT, HEADER_LOGO_TOP, width=HEADER_LOGO_WIDTH)

    add_text(slide, CONTENT_LEFT, EYEBROW_TOP, Inches(10), Inches(0.4), eyebrow.upper(), 15, GOLD)
    add_text(slide, CONTENT_LEFT, TITLE_TOP, CONTENT_WIDTH - Inches(2.6), TITLE_HEIGHT, title, 39, NAVY, bold=True)
    add_rect(slide, CONTENT_LEFT, HEADER_LINE_TOP, CONTENT_WIDTH, HEADER_LINE_HEIGHT, NAVY)

    add_rect(slide, CONTENT_LEFT, FOOTER_LINE_TOP, CONTENT_WIDTH, Inches(0.016), LIGHT_GRAY)
    add_text(slide, CONTENT_LEFT, FOOTER_TEXT_TOP, Inches(6), Inches(0.4), footer_text, 15, TEXT_GRAY)
    add_text(slide, CONTENT_RIGHT - Inches(1.982), FOOTER_TEXT_TOP, Inches(1.982), Inches(0.4),
             f"{page_num:02d}", 15, TEXT_GRAY, align=PP_ALIGN.RIGHT)

    return slide

def _bar_geometry(n, area_width, gap_ratio=BAR_GAP_RATIO):
    """bar_width, pitch for `n` bars spread across `area_width` with a
    consistent gap-to-bar-width ratio."""
    bar_width = area_width / (n + (n - 1) * gap_ratio)
    pitch = bar_width * (1 + gap_ratio)
    return bar_width, pitch


def _nice_ticks(max_value, target_count=4):
    """0, step, 2*step, ... up to a "nice" (1/2/5/10-multiple) round number
    at or above `max_value`, for a y-axis with human-friendly tick values."""
    if max_value <= 0:
        return [0, 1]
    raw_step = max_value / target_count
    magnitude = 10 ** math.floor(math.log10(raw_step))
    residual = raw_step / magnitude
    if residual > 5:
        step = 10 * magnitude
    elif residual > 2:
        step = 5 * magnitude
    elif residual > 1:
        step = 2 * magnitude
    else:
        step = magnitude
    ticks = [0]
    while ticks[-1] < max_value - 1e-9:
        ticks.append(round(ticks[-1] + step, 6))
    return ticks


def add_vertical_bar_chart(slide, categories, values, baseline, max_bar_height,
                            value_fmt=str, left=CONTENT_LEFT, width=CONTENT_WIDTH,
                            highlight_max=True, colors=None, value_label_size=19.5,
                            category_label_size=19.5, show_value_labels=True,
                            show_y_axis=True, bold_value_labels=False, bold_category_labels=False,
                            gap_ratio=BAR_GAP_RATIO):
    """A column chart drawn as native rectangles. Bars sit on `baseline`
    (their bottom edge) and grow upward; a "nice" round number at or above
    the largest value reaches `max_bar_height`, with that scale shown as a
    y-axis (gridlines + tick labels) to its left. `colors`, if given, is a
    list of one RGBColor per bar (overrides the default
    navy-with-gold-max-highlight)."""
    n = len(categories)
    if n == 0:
        return
    max_value = max(values) if values else 0
    y_axis_width = Inches(0.95) if show_y_axis else Inches(0)
    chart_left = left + y_axis_width
    chart_width = width - y_axis_width
    bar_width, pitch = _bar_geometry(n, chart_width, gap_ratio=gap_ratio)

    ticks = _nice_ticks(max_value)
    nice_max = ticks[-1]

    if show_y_axis:
        for t in ticks:
            y = baseline - int(max_bar_height * (t / nice_max))
            add_rect(slide, chart_left, y, chart_width, Inches(0.008), LIGHT_GRAY)
            add_text(slide, left, int(y - Inches(0.14)), y_axis_width - Inches(0.12), Inches(0.28),
                     value_fmt(t), 12, TEXT_GRAY, align=PP_ALIGN.RIGHT, anchor=MSO_ANCHOR.MIDDLE)

    for i, (cat, value) in enumerate(zip(categories, values)):
        bar_left = chart_left + int(i * pitch)
        bar_height = int(max_bar_height * (value / nice_max)) if nice_max else 0
        bar_top = baseline - bar_height

        if colors is not None:
            color = colors[i % len(colors)]
        elif highlight_max and max_value and value == max_value:
            color = GOLD
        else:
            color = NAVY
        add_rect(slide, bar_left, bar_top, bar_width, max(bar_height, 1), color)

        if show_value_labels:
            label_h = Inches(0.46)
            add_text(slide, bar_left - Inches(0.3), bar_top - label_h - Inches(0.08), bar_width + Inches(0.6),
                     label_h, value_fmt(value), value_label_size, TEXT_DARK, bold=bold_value_labels,
                     align=PP_ALIGN.CENTER)

        add_text(slide, bar_left - Inches(0.3), baseline + Inches(0.12), bar_width + Inches(0.6), Inches(0.46),
                 str(cat), category_label_size, TEXT_DARK, bold=bold_category_labels, align=PP_ALIGN.CENTER)


def add_hourly_panel(slide, left, top, width, height, day_label, date_label, total_label, hours, values,
                      value_fmt=lambda v: str(int(v))):
    """One mini bar-chart panel for the 'calls offered by hour' /
    'average wait time by hour' grid slides: a "Day: total" header, a
    y-axis (gridlines + tick labels) against a "nice" round scale, bars
    with per-bar value labels, every hour labeled below at a legible
    size (rotated vertical so a full 24-hour range never overlaps or
    shrinks), and the chart's calendar date centered underneath."""
    header_h = Inches(0.42)
    header_box = slide.shapes.add_textbox(left, top, width, header_h)
    tf = header_box.text_frame
    tf.word_wrap = False
    tf.margin_left = tf.margin_right = tf.margin_top = tf.margin_bottom = 0
    p = tf.paragraphs[0]
    p.alignment = PP_ALIGN.CENTER
    run1 = p.add_run()
    run1.text = f"{day_label}: "
    run1.font.name = FONT_NAME
    run1.font.size = Pt(24)
    run1.font.bold = True
    run1.font.color.rgb = NAVY
    run2 = p.add_run()
    run2.text = total_label
    run2.font.name = FONT_NAME
    run2.font.size = Pt(24)
    run2.font.bold = True
    run2.font.color.rgb = NAVY

    add_chart_border(slide, left, top, width, height)

    y_axis_width = Inches(0.68)
    chart_left = left + y_axis_width
    chart_width = width - y_axis_width

    axis_label_len = Inches(0.9)
    axis_label_gap = Inches(0.06)
    axis_label_thickness = Inches(0.34)
    date_gap = Inches(0.06)
    date_h = Inches(0.34)
    baseline = (top + height - date_gap - date_h - axis_label_gap
                - axis_label_len - Inches(0.05))
    value_label_h = Inches(0.28)
    value_label_gap = Inches(0.05)
    header_gap = Inches(0.08)
    max_bar_height = baseline - (top + header_h + header_gap) - value_label_h - value_label_gap

    n = len(hours)
    if n:
        max_value = max(values) if values else 0
        bar_width, pitch = _bar_geometry(n, chart_width, gap_ratio=0.22)
        ticks = _nice_ticks(max_value)
        nice_max = ticks[-1]

        for t in ticks:
            y = baseline - int(max_bar_height * (t / nice_max))
            add_rect(slide, chart_left, y, chart_width, Inches(0.008), LIGHT_GRAY)
            add_text(slide, left, int(y - Inches(0.14)), y_axis_width - Inches(0.1), Inches(0.28),
                     value_fmt(t), 13, TEXT_GRAY, align=PP_ALIGN.RIGHT, anchor=MSO_ANCHOR.MIDDLE)

        for i, (hour, value) in enumerate(zip(hours, values)):
            bar_left = chart_left + int(i * pitch)
            bar_height = int(max_bar_height * (value / nice_max)) if nice_max else 0
            bar_top = baseline - bar_height
            color = GOLD if max_value and value == max_value else NAVY
            add_rect(slide, bar_left, bar_top, bar_width, max(bar_height, 1), color)
            add_text(slide, bar_left - Inches(0.25), bar_top - value_label_h - value_label_gap,
                     bar_width + Inches(0.5), value_label_h, value_fmt(value), 15, TEXT_DARK,
                     bold=True, align=PP_ALIGN.CENTER)

            # Pre-rotation the box is wide and short (normal horizontal text
            # layout); rotating it -90 swaps that into a narrow vertical
            # strip, so its width/height here must be set for the
            # *unrotated* text, not the final on-slide footprint.
            bar_center = bar_left + bar_width / 2
            center_y = baseline + axis_label_gap + axis_label_len / 2
            add_text(slide, int(bar_center - axis_label_len / 2), int(center_y - axis_label_thickness / 2),
                     int(axis_label_len), int(axis_label_thickness), hour_label(hour), 18, TEXT_DARK,
                     bold=True, align=PP_ALIGN.CENTER, anchor=MSO_ANCHOR.MIDDLE, rotation=-90)

    add_rect(slide, chart_left, baseline, chart_width, Inches(0.01), LIGHT_GRAY)

    if date_label:
        add_text(slide, left, baseline + axis_label_gap + axis_label_len + date_gap, width, date_h,
                 date_label, 22, NAVY, bold=True, align=PP_ALIGN.CENTER)


def add_hourly_grid(slide, panels, top, bottom, left=CONTENT_LEFT, width=CONTENT_WIDTH):
    """Places up to 4 hourly panels in a 2-column grid. A row with fewer
    than 2 panels (e.g. Sunday alone) has its panel(s) widened to fill the
    full row instead of sitting at a fixed half-width, so a lone panel is
    a perfect full-width fit rather than a centered half-width box.
    `left`/`width` default to the standard content margins but can be
    widened for slides made up entirely of charts."""
    n = len(panels)
    ncols = 2
    nrows = (n + ncols - 1) // ncols
    gutter = Inches(0.08)
    panel_height = (bottom - top - gutter * (nrows - 1)) / nrows

    for row in range(nrows):
        row_panels = panels[row * ncols:(row + 1) * ncols]
        ncols_this_row = len(row_panels)
        panel_width = (width - gutter * (ncols_this_row - 1)) / ncols_this_row
        panel_top = top + row * (panel_height + gutter)
        for col, panel_kwargs in enumerate(row_panels):
            panel_left = left + col * (panel_width + gutter)
            add_hourly_panel(slide, panel_left, panel_top, panel_width, panel_height, **panel_kwargs)


def add_horizontal_bar_list(slide, items, values, top):
    """The 'Top 10 Primary Configuration Items' leaderboard: item name on
    the left, a horizontal bar scaled to the max value, the number after
    the bar. Gold-highlights the top (#1) item."""
    n = len(items)
    if n == 0:
        return
    max_value = max(values)
    label_width = Inches(6.0)
    value_width = Inches(0.8)
    gap = Inches(0.25)
    bar_area_left = CONTENT_LEFT + label_width + gap
    bar_area_width = CONTENT_WIDTH - label_width - gap - value_width - gap
    row_height = Inches(0.63)
    bar_height = Inches(0.4)

    for i, (item, value) in enumerate(zip(items, values)):
        row_top = top + int(i * row_height)
        add_text(slide, CONTENT_LEFT, row_top, label_width, row_height, str(item), 20, NAVY, bold=True,
                 anchor=MSO_ANCHOR.MIDDLE)
        bar_width = int(bar_area_width * (value / max_value)) if max_value else 0
        color = GOLD if value == max_value else NAVY
        bar_top = row_top + int((row_height - bar_height) / 2)
        add_rect(slide, bar_area_left, bar_top, max(bar_width, 1), bar_height, color)
        add_text(slide, bar_area_left + bar_width + gap, row_top, value_width, row_height, str(value), 18, NAVY,
                 bold=True, anchor=MSO_ANCHOR.MIDDLE)


def add_gauge(slide, left, top, width, value, max_value=5):
    """The 'Overall Satisfaction Rate' horizontal gauge: a light track, a
    fill proportional to value/max_value (green at/above
    GAUGE_GOOD_THRESHOLD, red below it), tick labels 0..max_value, and the
    value itself shown large above the gauge."""
    add_text(slide, left, top, width, Inches(1.0), f"{value:.2f}", 48, NAVY, bold=True, align=PP_ALIGN.CENTER)

    gauge_top = top + Inches(1.2)
    gauge_height = Inches(0.583)
    add_rect(slide, left, gauge_top, width, gauge_height, GAUGE_BG)
    fill_width = int(width * (value / max_value)) if max_value else 0
    fill_color = GREEN if value >= GAUGE_GOOD_THRESHOLD else RED
    add_rect(slide, left, gauge_top, max(fill_width, 1), gauge_height, fill_color)

    ticks_top = gauge_top + gauge_height + Inches(0.15)
    for i in range(max_value + 1):
        x = left + int(width * i / max_value) - Inches(0.11)
        add_text(slide, x, ticks_top, Inches(0.4), Inches(0.4), str(i), 18, TEXT_GRAY, align=PP_ALIGN.CENTER)

    add_text(slide, left, ticks_top + Inches(0.5), width, Inches(0.4), "Rating out of 5", 18, TEXT_GRAY,
             align=PP_ALIGN.CENTER)


def add_statement(slide, text, top=Inches(3.6)):
    """The No CI Found streak sentence: a short gold accent bar above a
    large bold navy statement."""
    add_rect(slide, CONTENT_LEFT, top, Inches(1.65), Inches(0.042), GOLD)
    add_text(slide, CONTENT_LEFT, top + Inches(0.6), Inches(16), Inches(2.5), text, 40, NAVY, bold=True,
             word_wrap=True)

def _fit_table_rows(n_rows, header_height, max_row_height=Inches(0.63), min_row_height=Inches(0.38),
                     min_font_size=8):
    """Row height (and a matching font size) that fits `n_rows` between
    the header and the footer line, capped so a short table doesn't
    stretch its rows unnaturally tall. Callers are expected to keep
    `n_rows` at or under `_rows_per_page()` for the same header_height/
    min_row_height, so this never has to shrink rows past readability
    (and off the bottom of the slide) to fit everything on one page --
    unless `min_row_height`/`min_font_size` are deliberately lowered to
    force everything onto a single slide no matter the row count (see
    build_pivot_table_slide)."""
    available = FOOTER_LINE_TOP - Inches(0.15) - CONTENT_TOP - header_height
    row_height = available / n_rows if n_rows else max_row_height
    row_height = max(min(row_height, max_row_height), min_row_height)
    font_size = max(min_font_size, min(18, 18 * row_height / Inches(0.63)))
    return int(row_height), font_size


def _rows_per_page(header_height, min_row_height=Inches(0.38)):
    """How many table rows fit on one slide without shrinking below
    min_row_height -- the pagination unit for a table that might have
    more records than one slide can hold."""
    available = FOOTER_LINE_TOP - Inches(0.15) - CONTENT_TOP - header_height
    return max(1, int(available / min_row_height))


def _paginate_table(df, header_height):
    """Splits `df` into as many same-sized chunks as needed so each one
    fits on a single slide (see _rows_per_page). A table with "a lot of
    records" spills onto additional slides instead of running off the
    bottom of one."""
    if len(df) == 0:
        return [df]
    rows_per_page = _rows_per_page(header_height)
    return [df.iloc[i:i + rows_per_page] for i in range(0, len(df), rows_per_page)]


def build_pivot_table_slide(slide, pivot_df):
    headers = ["Primary configuration item", "Summary", "Tickets"]
    col_widths = [Inches(4.9), Inches(11.7), Inches(1.55)]
    header_height = Inches(0.73)
    left = CONTENT_LEFT
    top = CONTENT_TOP
    # Always exactly one slide: no row-height/font floor, so a heavy week
    # shrinks to fit rather than being paginated or running off the slide.
    row_height, font_size = _fit_table_rows(len(pivot_df), header_height,
                                             min_row_height=Inches(0), min_font_size=4)
    font_size = min(font_size + 2, 20)

    add_rect(slide, left, top, CONTENT_WIDTH, header_height, NAVY)
    col_x = []
    x = left
    for width, header in zip(col_widths, headers):
        col_x.append(x)
        add_text(slide, x + Inches(0.2), top, width - Inches(0.3), header_height, header, 18, WHITE, bold=True,
                 anchor=MSO_ANCHOR.MIDDLE)
        x += width

    row_top = top + header_height
    last_item = None
    group_index = -1
    for _, row in pivot_df.reset_index(drop=True).iterrows():
        item = str(row["Primary Configuration Item"])
        is_new_group = item != last_item
        if is_new_group:
            group_index += 1
            last_item = item
        if group_index % 2 == 1:
            add_rect(slide, left, row_top, CONTENT_WIDTH, row_height, ROW_SHADE)
        add_rect(slide, left, row_top, CONTENT_WIDTH, Inches(0.01), LIGHT_GRAY)

        add_text(slide, col_x[0] + Inches(0.2), row_top, col_widths[0] - Inches(0.3), row_height,
                 item if is_new_group else "", font_size, NAVY, bold=True, anchor=MSO_ANCHOR.MIDDLE)
        add_text(slide, col_x[1] + Inches(0.2), row_top, col_widths[1] - Inches(0.3), row_height,
                 str(row["Summary"]), font_size, TEXT_DARK, bold=True, anchor=MSO_ANCHOR.MIDDLE)
        add_text(slide, col_x[2] + Inches(0.2), row_top, col_widths[2] - Inches(0.3), row_height,
                 str(row["Number"]), font_size, NAVY, bold=True, anchor=MSO_ANCHOR.MIDDLE)
        row_top += row_height


def build_records_table_slide(slide, df):
    n_cols = len(df.columns)
    col_width = CONTENT_WIDTH / n_cols
    col_widths = [col_width] * n_cols
    header_height = Inches(0.73)
    left = CONTENT_LEFT
    top = CONTENT_TOP
    row_height, font_size = _fit_table_rows(len(df), header_height)

    add_rect(slide, left, top, CONTENT_WIDTH, header_height, NAVY)
    col_x = []
    x = left
    for width, header in zip(col_widths, df.columns):
        col_x.append(x)
        add_text(slide, x + Inches(0.15), top, width - Inches(0.25), header_height, str(header),
                 min(13, font_size), WHITE, bold=True, anchor=MSO_ANCHOR.MIDDLE)
        x += width

    row_top = top + header_height
    for i, (_, row) in enumerate(df.reset_index(drop=True).iterrows()):
        if i % 2 == 1:
            add_rect(slide, left, row_top, CONTENT_WIDTH, row_height, ROW_SHADE)
        add_rect(slide, left, row_top, CONTENT_WIDTH, Inches(0.01), LIGHT_GRAY)
        for col, value in enumerate(row):
            text = "" if pd.isna(value) else str(value)
            add_text(slide, col_x[col] + Inches(0.15), row_top, col_widths[col] - Inches(0.25), row_height, text,
                     min(11, font_size), TEXT_DARK, anchor=MSO_ANCHOR.MIDDLE)
        row_top += row_height

def build_cover_slide(prs, blank_layout, start_date, end_date, logo_path):
    """The title slide: a short navy rule, an eyebrow label, a big date-range
    headline, a reporting-period subtitle, and a footer rule + label. Carries
    no page number (matching the reference design)."""
    slide = prs.slides.add_slide(blank_layout)

    if logo_path is not None:
        slide.shapes.add_picture(str(logo_path), HEADER_LOGO_LEFT, HEADER_LOGO_TOP, width=HEADER_LOGO_WIDTH)

    add_rect(slide, CONTENT_LEFT, Inches(1.573), CONTENT_WIDTH, Inches(0.042), NAVY)
    add_text(slide, CONTENT_LEFT, Inches(1.917), CONTENT_WIDTH, Inches(0.4),
             "WEEKLY OPERATIONS REPORT", 15, COVER_EYEBROW_COLOR, bold=True)
    add_text(slide, CONTENT_LEFT, Inches(4.0), CONTENT_WIDTH, Inches(2.4),
             f"HAL Weekly Report {cover_date_range(start_date, end_date)}", 58, NAVY, bold=True,
             word_wrap=True)
    add_text(slide, CONTENT_LEFT, Inches(7.125), CONTENT_WIDTH, Inches(0.6),
             f"Reporting period {format_date(start_date)} – {format_date(end_date)}", 30, TEXT_DARK)

    add_rect(slide, CONTENT_LEFT, Inches(9.542), CONTENT_WIDTH, Inches(0.016), LIGHT_GRAY)
    add_text(slide, CONTENT_LEFT, Inches(9.781), CONTENT_WIDTH, Inches(0.4),
             "SERVICE DESK — HOLLAND AMERICA LINE", 15, TEXT_GRAY)

    return slide

def build_presentation(report_title, start_date, end_date, calls, hours, apps, pivot_df,
                        noci_text, noci_df, csat_votes, csat_overall, output_path, logo_path=None):
    prs = Presentation()
    prs.slide_width = SLIDE_WIDTH
    prs.slide_height = SLIDE_HEIGHT
    blank_layout = prs.slide_layouts[6]

    build_cover_slide(prs, blank_layout, start_date, end_date, logo_path)

    footer_text = f"HAL weekly report · {format_date_range(start_date, end_date)}"
    page_num = 0

    def add_slide(eyebrow, title):
        nonlocal page_num
        page_num += 1
        return new_slide(prs, blank_layout, eyebrow, title, footer_text, page_num, logo_path)

    # Slide 1: weekly title + calls per weekday
    slide = add_slide("Call Volume", report_title)
    counts = calls["Weekday"].value_counts().reindex(WEEKDAYS, fill_value=0)
    add_vertical_bar_chart(slide, list(counts.index), list(counts.values), baseline=Inches(9.37),
                           max_bar_height=Inches(6.3), gap_ratio=0.8, value_fmt=lambda v: str(int(v)),
                           value_label_size=22, category_label_size=22,
                           bold_value_labels=True, bold_category_labels=True)

    # 2 panels per slide (instead of 4) so each hourly panel gets roughly
    # double the vertical room: Mon+Tue, Wed+Thu, Fri+Sat, Sun (alone, centered).
    weekday_groups = [WEEKDAYS[i:i + 2] for i in range(0, len(WEEKDAYS), 2)]

    # Slides 2-5: HAL Calls Offered by Hour
    for i, day_group in enumerate(weekday_groups):
        slide = add_slide("Call Volume", f"HAL Calls Offered by Hour ({i + 1} of {len(weekday_groups)})")
        panels = []
        for day in day_group:
            day_df = calls[calls["Weekday"] == day]
            values = day_df.groupby("Hour").size().reindex(hours, fill_value=0)
            chart_date = date_for_weekday(start_date, end_date, day)
            panels.append(dict(
                day_label=day,
                date_label=format_day_month(chart_date) if chart_date is not None else "",
                total_label=f"{len(day_df)} calls",
                hours=hours,
                values=list(values.values),
                value_fmt=lambda v: str(int(v)),
            ))
        add_hourly_grid(slide, panels, top=Inches(2.15), bottom=Inches(10.35), left=HOURLY_LEFT, width=HOURLY_WIDTH)

    # Slides 6-9: HAL Average Waiting Time
    for i, day_group in enumerate(weekday_groups):
        slide = add_slide("Call Volume", f"HAL Average Waiting Time ({i + 1} of {len(weekday_groups)})")
        panels = []
        for day in day_group:
            day_df = calls[calls["Weekday"] == day]
            values = day_df.groupby("Hour")["TotalQueueSeconds"].mean().reindex(hours).fillna(0)
            day_avg = day_df["TotalQueueSeconds"].mean()
            chart_date = date_for_weekday(start_date, end_date, day)
            panels.append(dict(
                day_label=day,
                date_label=format_day_month(chart_date) if chart_date is not None else "",
                total_label=f"Avg {seconds_to_mmss(day_avg)}",
                hours=hours,
                values=list(values.values),
                value_fmt=seconds_to_mmss,
            ))
        add_hourly_grid(slide, panels, top=Inches(2.15), bottom=Inches(10.35), left=HOURLY_LEFT, width=HOURLY_WIDTH)

    # Slide 6: HAL Top Cases (Top 10 leaderboard)
    slide = add_slide("Applications", "HAL Top Cases")
    add_horizontal_bar_list(slide, list(apps.index), list(apps.values), top=CONTENT_TOP)

    # Slide 7: Ticket detail by configuration item (pivot table). Always one
    # slide -- build_pivot_table_slide shrinks rows/font with no floor on a
    # heavy week rather than paginating.
    slide = add_slide("Applications", "Ticket detail by configuration item")
    build_pivot_table_slide(slide, pivot_df)

    # Slide 8+: HAL No CI Found (streak statement, or the actual records --
    # also paginated -- if the streak broke)
    if noci_text is not None:
        slide = add_slide("Applications", "HAL No CI Found")
        add_statement(slide, noci_text)
    else:
        noci_pages = _paginate_table(noci_df, header_height=Inches(0.73))
        for i, page_df in enumerate(noci_pages):
            suffix = f" ({i + 1} of {len(noci_pages)})" if len(noci_pages) > 1 else ""
            slide = add_slide("Applications", f"HAL No CI Found{suffix}")
            build_records_table_slide(slide, page_df)

    # Slide 9: C-SAT results (summary line, Overall satisfaction rate on the
    # left, Votes by response on the right)
    slide = add_slide("Customer Satisfaction", "C-SAT results")
    overall_label = csat_overall[0][0] if csat_overall[0] else ""
    overall_value = csat_overall[1][0][1][0] if csat_overall[1] and csat_overall[1][0][1] else 0
    add_text(slide, CONTENT_LEFT, Inches(2.667), CONTENT_WIDTH, Inches(0.5),
             f"Overall satisfaction of {overall_value:.2f} out of 5 for {format_month_year(start_date)}",
             25.5, TEXT_DARK, bold=True)

    section_top = Inches(3.4)
    half_width = (CONTENT_WIDTH - Inches(0.6)) / 2
    add_text(slide, CONTENT_LEFT, section_top, half_width, Inches(0.6), "Overall satisfaction rate", 22.5, NAVY,
             bold=True)
    add_gauge(slide, CONTENT_LEFT, section_top + Inches(1.0), half_width, overall_value, max_value=5)

    votes_left = CONTENT_LEFT + half_width + Inches(0.6)
    votes_labels = [clean_vote_label(name) for name, _ in csat_votes[1]]
    votes_values = [vals[0] if vals else 0 for _, vals in csat_votes[1]]
    add_text(slide, votes_left, section_top, half_width, Inches(0.6), "Votes by response", 22.5, NAVY, bold=True)
    add_vertical_bar_chart(slide, votes_labels, votes_values, baseline=Inches(8.9), max_bar_height=Inches(4.3),
                           value_fmt=lambda v: str(int(v)), left=votes_left, width=half_width,
                           colors=vote_colors(votes_labels), value_label_size=16, category_label_size=15,
                           bold_value_labels=True, bold_category_labels=True)

    prs.save(output_path)

def main():
    parser = argparse.ArgumentParser(description="Generate the HAL weekly PPTX report.")
    parser.add_argument("--csv", default=str(DEFAULT_CSV_PATH),
                         help="Path to the calls CSV export, or a folder containing it "
                              f"(default: {DEFAULT_CSV_PATH}).")
    parser.add_argument("--excel", default=str(DEFAULT_EXCEL_PATH),
                         help=f"Path to the tickets Excel export (default: {DEFAULT_EXCEL_PATH}).")
    parser.add_argument("--output", default=None,
                         help="Output .pptx path (default: "
                              f"'{DEFAULT_OUTPUT_PATH / 'HAL Weekly Report - <Start Date> to <End Date>.pptx'}', "
                              "built from the dates you enter).")
    parser.add_argument("--logo", default=str(DEFAULT_LOGO_PATH),
                         help=f"Logo image placed in the top-right corner of every slide "
                              f"(default: {DEFAULT_LOGO_PATH}). Skipped if not found.")
    parser.add_argument("--noci", default=str(DEFAULT_NOCI_PATH),
                         help="Path to the No CI Found Excel export, used only when you answer "
                              f"'no' to the No CI Found prompt (default: {DEFAULT_NOCI_PATH}).")
    args = parser.parse_args()

    excel_path = Path(args.excel)
    logo_path = Path(args.logo)
    if not logo_path.is_file():
        print(f"Logo not found at {logo_path}, skipping it.")
        logo_path = None
    csv_path = resolve_csv_path(args.csv)
    if not csv_path.exists():
        raise FileNotFoundError(f"CSV file not found: {csv_path}")
    if not excel_path.exists():
        raise FileNotFoundError(f"Excel file not found: {excel_path}")
    if excel_path.is_dir():
        raise IsADirectoryError(
            f"--excel points at a folder, not a file: {excel_path}\n"
            "Add the actual Excel filename to the end of the path."
        )
    csat_path = resolve_csat_path(csv_path.parent)
    if csat_path is None:
        raise FileNotFoundError(
            f"No C-SAT file (an .xlsx with 'C-SAT' in its name) found in {csv_path.parent}"
        )

    start_date, end_date, report_title = get_date_inputs()

    no_ci_found_this_week = prompt_yes_no("No CI Found This Week")
    if no_ci_found_this_week:
        streak = load_streak() + 1
        save_streak(streak)
        noci_text = f'For the {ordinal(streak)} straight week, there were no "No CI Found"'
        noci_df = None
    else:
        save_streak(0)
        noci_path = Path(args.noci)
        if not noci_path.exists():
            raise FileNotFoundError(f"No CI Found file not found: {noci_path}")
        noci_df = load_noci_records(noci_path)
        noci_text = None

    if args.output is not None:
        output_path = Path(args.output)
        if output_path.is_dir() or not output_path.suffix:
            raise IsADirectoryError(
                f"--output points at a folder, not a file: {output_path}\n"
                "Add a .pptx filename to the end of the path."
            )
    else:
        output_path = DEFAULT_OUTPUT_PATH / f"HAL Weekly Report - {format_date(start_date)} to {format_date(end_date)}.pptx"

    calls = load_calls(csv_path, start_date, end_date)
    apps = load_apps(excel_path)
    pivot_df = load_pivot_table(excel_path)
    hours = _hour_range(calls)

    votes = extract_named_chart_data(csat_path, CSAT_SHEET_NAME, CSAT_VOTES_CHART)
    overall = extract_named_chart_data(csat_path, CSAT_SHEET_NAME, CSAT_OVERALL_CHART)

    build_presentation(
        report_title,
        start_date,
        end_date,
        calls,
        hours,
        apps,
        pivot_df,
        noci_text,
        noci_df,
        votes,
        overall,
        output_path,
        logo_path=logo_path,
    )

    print(f"Report saved to {output_path}")


if __name__ == "__main__":
    main()
