"""
HAL Weekly Report generator — restyled edition.

Same data pipeline as before (CSV of call records, SNOW ticket export,
No CI Found export, C-SAT workbook), but the presentation layer has been
rebuilt around a single design system instead of python-pptx / matplotlib
defaults:

  * a real cover slide and an "At a glance" numbers slide for management
  * numbered section dividers in Holland America navy
  * every content slide on one grid: kicker, flush-left title, 2pt rule,
    footer with the reporting period and a page number
  * charts drawn on transparent grounds in the brand palette, with the
    peak bar picked out in gold, no chart junk, no bold-everything
  * tables with a navy header, hairline rules and tinted group bands
    instead of PowerPoint's default blue banding

Everything is still pure python-pptx + matplotlib (no new dependencies),
and all slide text is real PowerPoint text, so it stays editable.

    python generate_report.py [--csv PATH] [--excel PATH] [--output PATH] [--noci PATH]

If --csv/--excel/--noci are omitted, the DEFAULT_* paths below are used
(edit them to match the actual filenames each week). If --output is
omitted, the file is named
"HAL Weekly Report - <Start Date> to <End Date>.pptx" and saved inside
the DEFAULT_OUTPUT_PATH folder, using the dates you enter below.

The script then interactively asks for the same six values the original
script used: Start Day Date, End Day Date, Start Date Month, End Date Month,
Start Date Year, End Date Year, plus a No CI Found This Week (yes/no)
prompt. Answering "yes" increments a streak counter persisted in
STREAK_STATE_PATH and shows it in a sentence; answering "no" resets the
streak and lists the actual records from the No CI Found file instead.
"""

def parse_month(value):
    value = value.strip()
    if value.isdigit():
        month = int(value)
        if 1 <= month <= 12:
            return month
        raise ValueError(f"Month number out of range: {value}")
    for fmt_names in (calendar.month_name, calendar.month_abbr):
        for i, name in enumerate(fmt_names):
            if name and name.lower() == value.lower():
                return i
    raise ValueError(f"Could not parse month: {value}")


def prompt_int(label):
    while True:
        raw = input(f"{label}: ").strip()
        if raw.isdigit():
            return int(raw)
        print("Please enter a number.")


def prompt_month(label):
    while True:
        raw = input(f"{label}: ").strip()
        try:
            return parse_month(raw)
        except ValueError as exc:
            print(exc)


def prompt_yes_no(label):
    while True:
        raw = input(f"{label} (yes/no): ").strip().lower()
        if raw in ("y", "yes"):
            return True
        if raw in ("n", "no"):
            return False
        print("Please enter yes or no.")

def resolve_csv_path(path):
    """If `path` is a folder, returns the .csv file inside it (the most
    recently modified one, if there's more than one). Returns `path`
    unchanged if it's already a file."""
    path = Path(path)
    if not path.is_dir():
        return path
    csv_files = sorted(path.glob("*.csv"), key=lambda p: p.stat().st_mtime, reverse=True)
    if not csv_files:
        raise FileNotFoundError(f"No .csv file found in folder: {path}")
    if len(csv_files) > 1:
        names = ", ".join(p.name for p in csv_files)
        print(f"Multiple CSV files found in {path} ({names}); using the most recently modified: {csv_files[0].name}")
    return csv_files[0]


def resolve_csat_path(folder):
    """Finds the .xlsx file whose name contains "C-SAT" (any spacing/case)
    inside `folder`. Returns None if no such file is found."""
    folder = Path(folder)

    def norm(s):
        return re.sub(r"[^a-z0-9]", "", s.lower())

    matches = [p for p in folder.glob("*.xlsx") if "csat" in norm(p.stem)]
    if not matches:
        return None
    if len(matches) > 1:
        matches = sorted(matches, key=lambda p: p.stat().st_mtime, reverse=True)
        names = ", ".join(p.name for p in matches)
        print(f"Multiple C-SAT files found in {folder} ({names}); using the most recently modified: {matches[0].name}")
    return matches[0]


def load_calls(csv_path, start_date, end_date):
    df = pd.read_csv(csv_path)

    df["ParsedDate"] = pd.to_datetime(df["Date"], format="%m/%d/%y %I:%M %p", errors="coerce")
    df = df.dropna(subset=["ParsedDate"])

    # Only count calls that landed in one of the HAL queues.
    df = df[df["Queue"].isin(VALID_QUEUES)].copy()

    end_of_day = end_date + pd.Timedelta(hours=23, minutes=59, seconds=59)
    df = df[(df["ParsedDate"] >= start_date) & (df["ParsedDate"] <= end_of_day)]

    df["Weekday"] = pd.Categorical(df["ParsedDate"].dt.day_name(), categories=WEEKDAYS, ordered=True)
    df["Hour"] = df["ParsedDate"].dt.hour

    total_queue = df["Total Queue"].astype(str).str.strip()
    df["TotalQueueSeconds"] = pd.to_timedelta(total_queue, errors="coerce").dt.total_seconds()

    return df


_EXCEL_CACHE = {}


def read_tickets(xlsx_path):
    """Reads (and caches) the SNOW ticket export, so the KPI slide, the
    applications chart and the pivot table share one pass over the file."""
    key = str(xlsx_path)
    if key not in _EXCEL_CACHE:
        _EXCEL_CACHE[key] = pd.read_excel(xlsx_path)
    return _EXCEL_CACHE[key]


def load_apps(xlsx_path):
    df = read_tickets(xlsx_path)
    items = df["Primary Configuration Item"].dropna().astype(str)
    items = items.apply(lambda s: re.sub(r"[​‌‍﻿]", "", s).strip())
    items = items[items != ""]
    return items.value_counts().head(TOP_N_APPS)


def load_ticket_total(xlsx_path):
    return len(read_tickets(xlsx_path))


def _clean_text(series):
    return series.fillna("").astype(str).apply(lambda s: re.sub(r"[​‌‍﻿]", "", s).strip())


def load_pivot_table(xlsx_path):
    """Primary Configuration Item / Summary / Number of records, for the
    fixed PIVOT_ITEMS whitelist, sorted by item then by count descending."""
    df = read_tickets(xlsx_path)
    df = df.assign(
        **{"Primary Configuration Item": _clean_text(df["Primary Configuration Item"])}
    )
    df = df[df["Primary Configuration Item"].isin(PIVOT_ITEMS)].copy()
    df["Summary"] = _clean_text(df["Summary"])
    df = df[df["Summary"] != ""]

    pivot = (
        df.groupby(["Primary Configuration Item", "Summary"])
        .size()
        .reset_index(name="Number")
    )
    pivot["Primary Configuration Item"] = pd.Categorical(
        pivot["Primary Configuration Item"], categories=PIVOT_ITEMS, ordered=True
    )
    pivot = pivot.sort_values(
        ["Primary Configuration Item", "Number"], ascending=[True, False]
    ).reset_index(drop=True)
    return pivot


def load_noci_records(xlsx_path):
    """Raw "No CI Found" ticket records, with text columns cleaned and the
    Opened column formatted as a readable date string."""
    df = pd.read_excel(xlsx_path)
    for col in df.columns:
        if df[col].dtype == object:
            df[col] = _clean_text(df[col])
    if "Opened" in df.columns:
        opened = pd.to_datetime(df["Opened"], errors="coerce")
        df["Opened"] = opened.dt.strftime("%d %b %Y %H:%M").fillna(df["Opened"].astype(str))
    return df


# --------------------------------------------------------------------------- #
# C-SAT chart extraction (reads the named Excel chart objects' own cell
# references, so it stays correct even if the underlying data rows shift)
# --------------------------------------------------------------------------- #

_NS_R = "http://schemas.openxmlformats.org/officeDocument/2006/relationships"
_NS_MAIN = "http://schemas.openxmlformats.org/spreadsheetml/2006/main"
_NS_XDR = "http://schemas.openxmlformats.org/drawingml/2006/spreadsheetDrawing"
_NS_CHART_REL = "http://schemas.openxmlformats.org/drawingml/2006/chart"


def _get_named_charts(xlsx_path, sheet_name):
    """Returns {chart_name: openpyxl_chart_object} for every chart on
    `sheet_name`, keyed by Excel's own internal chart object name (e.g.
    "Chart 5"), by cross-referencing the sheet's drawing XML."""
    wb = openpyxl.load_workbook(xlsx_path, data_only=True)
    ws = wb[sheet_name]
    charts = ws._charts
    if not charts:
        return {}

    with zipfile.ZipFile(xlsx_path) as z:
        wb_xml = ET.fromstring(z.read("xl/workbook.xml"))
        sheet_rid = None
        for sheet_el in wb_xml.find(f"{{{_NS_MAIN}}}sheets"):
            if sheet_el.get("name") == sheet_name:
                sheet_rid = sheet_el.get(f"{{{_NS_R}}}id")
                break
        if sheet_rid is None:
            return {}

        wb_rels = ET.fromstring(z.read("xl/_rels/workbook.xml.rels"))
        sheet_target = next((rel.get("Target") for rel in wb_rels if rel.get("Id") == sheet_rid), None)
        if sheet_target is None:
            return {}
        sheet_target = sheet_target.lstrip("/")
        if not sheet_target.startswith("xl/"):
            sheet_target = "xl/" + sheet_target

        sheet_dir, sheet_file = sheet_target.rsplit("/", 1)
        sheet_rels_path = f"{sheet_dir}/_rels/{sheet_file}.rels"
        if sheet_rels_path not in z.namelist():
            return {}
        sheet_rels = ET.fromstring(z.read(sheet_rels_path))
        drawing_target = next((rel.get("Target") for rel in sheet_rels if "drawing" in rel.get("Type", "")), None)
        if drawing_target is None:
            return {}
        drawing_target = drawing_target.replace("../", "xl/")
        if not drawing_target.startswith("xl/"):
            drawing_target = "xl/drawings/" + drawing_target.rsplit("/", 1)[-1]

        drawing_xml = ET.fromstring(z.read(drawing_target))
        name_rid_pairs = []
        for gf in drawing_xml.iter(f"{{{_NS_XDR}}}graphicFrame"):
            cnvpr = gf.find(f".//{{{_NS_XDR}}}nvGraphicFramePr/{{{_NS_XDR}}}cNvPr")
            chart_el = gf.find(f".//{{{_NS_CHART_REL}}}chart")
            if cnvpr is not None and chart_el is not None:
                name_rid_pairs.append((cnvpr.get("name"), chart_el.get(f"{{{_NS_R}}}id")))

        drawing_dir, drawing_file = drawing_target.rsplit("/", 1)
        drawing_rels_path = f"{drawing_dir}/_rels/{drawing_file}.rels"
        drawing_rels = ET.fromstring(z.read(drawing_rels_path))
        rid_to_target = {rel.get("Id"): rel.get("Target") for rel in drawing_rels}

        ordered_names = [name for name, _ in name_rid_pairs]

    if len(ordered_names) != len(charts):
        return {}
    return dict(zip(ordered_names, charts))


def _resolve_ref_values(wb, ref):
    """Resolves a chart cell reference like "Summary!$B$8" or
    "Summary!$B$8:$F$8" into a flat list of live cell values."""
    sheet_part, cell_part = ref.split("!")
    sheet_part = sheet_part.strip("'")
    ws = wb[sheet_part]
    cell_part = cell_part.replace("$", "")
    cells = ws[cell_part]
    if isinstance(cells, tuple):
        flat = []
        for row in cells:
            if isinstance(row, tuple):
                flat.extend(c.value for c in row)
            else:
                flat.append(row.value)
        return flat
    return [cells.value]


def extract_named_chart_data(xlsx_path, sheet_name, chart_name):
    """Returns (categories, series) for the Excel chart object named
    `chart_name` on `sheet_name`, where series is a list of
    (series_name, [values]) using the chart's own live cell references."""
    wb = openpyxl.load_workbook(xlsx_path, data_only=True)
    named_charts = _get_named_charts(xlsx_path, sheet_name)
    chart = named_charts.get(chart_name)
    if chart is None:
        raise ValueError(f"Chart named '{chart_name}' not found on sheet '{sheet_name}'")

    categories = None
    series_data = []
    for s in chart.series:
        series_name = None
        if s.tx is not None and s.tx.strRef is not None:
            names = _resolve_ref_values(wb, s.tx.strRef.f)
            series_name = str(names[0]) if names else None
        values = [v if v is not None else 0 for v in _resolve_ref_values(wb, s.val.numRef.f)]
        if categories is None and s.cat is not None:
            if s.cat.strRef is not None:
                categories = [str(v) for v in _resolve_ref_values(wb, s.cat.strRef.f)]
            elif s.cat.numRef is not None:
                categories = [str(v) for v in _resolve_ref_values(wb, s.cat.numRef.f)]
        series_data.append((series_name, values))

    if categories is None:
        categories = [str(i) for i in range(len(series_data[0][1]))] if series_data else []
    return categories, series_data

def seconds_to_mmss(seconds):
    if pd.isna(seconds):
        seconds = 0
    minutes, sec = divmod(int(round(seconds)), 60)
    return f"{minutes:02d}:{sec:02d}"


def hour_label(hour):
    return pd.Timestamp(year=2000, month=1, day=1, hour=hour).strftime("%I %p").lstrip("0")


def ordinal(n):
    if 10 <= n % 100 <= 20:
        suffix = "th"
    else:
        suffix = {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")
    return f"{n}{suffix}"

def format_date(date):
    return f"{date.day} {calendar.month_abbr[date.month]} {date.year}"


def format_date_range(start_date, end_date):
    if start_date.date() == end_date.date():
        return format_date(start_date)
    return f"{format_date(start_date)} - {format_date(end_date)}"


def date_for_weekday(start_date, end_date, day):
    """First calendar date within [start_date, end_date] whose weekday name is `day`."""
    d = start_date
    while d <= end_date:
        if d.day_name() == day:
            return d
        d += pd.Timedelta(days=1)
    return None

def _style_axes(ax, tick_size, y_formatter=None):
    """Open frame: baseline only, light horizontal gridlines, no top/right
    spines, no tick marks."""
    ax.set_axisbelow(True)
    ax.grid(axis="y", color=GRID_HEX, linestyle="-", linewidth=0.8)
    ax.grid(axis="x", visible=False)
    for side in ("top", "right", "left"):
        ax.spines[side].set_visible(False)
    ax.spines["bottom"].set_color(RULE_HEX_CSS)
    ax.spines["bottom"].set_linewidth(1.0)
    ax.tick_params(axis="both", length=0, labelsize=tick_size)
    if y_formatter is not None:
        ax.yaxis.set_major_formatter(FuncFormatter(y_formatter))

def _bar_colors(values):
    """Navy bars, with the single peak bar in gold."""
    if len(values) == 0:
        return []
    peak = max(values)
    return [GOLD_HEX if (v == peak and peak > 0) else NAVY_HEX for v in values]

def make_single_hour_chart(df, day, hours, out_path, mode, chart_date=None):

    if mode == "count":
        values = day_df.groupby("Hour").size().reindex(hours, fill_value=0)
        headline = f"{len(day_df)} calls"
    else:
        values = day_df.groupby("Hour")["TotalQueueSeconds"].mean().reindex(hours).fillna(0)
        headline = f"avg {seconds_to_mmss(day_df['TotalQueueSeconds'].mean())}"

    labels = [hour_label(h) for h in hours]

    fig, ax = plt.subplots(figsize=(7.0, 3.9))
    y_fmt = None if mode == "count" else (lambda v, _: seconds_to_mmss(v))

    if mode == "count":
        bar_labels = [str(int(v)) if v > 0 else "" for v in values.values]
    else:
        bar_labels = [seconds_to_mmss(v) if v > 0 else "" for v in values.values]

    ax.margins(y=0.16)

def add_rect(slide, left, top, width, height, color):
    shape = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, left, top, width, height)
    shape.fill.solid()
    shape.fill.fore_color.rgb = color
    shape.line.fill.background()
    shape.shadow.inherit = False
    return shape


def add_text(slide, left, top, width, height, text, size, color,
             font=None, bold=False, spacing=None, align=PP_ALIGN.LEFT,
             anchor=MSO_ANCHOR.TOP, line_spacing=None, caps=False):
    box = slide.shapes.add_textbox(left, top, width, height)
    tf = box.text_frame
    tf.word_wrap = True
    tf.margin_left = 0
    tf.margin_right = 0
    tf.margin_top = 0
    tf.margin_bottom = 0
    tf.vertical_anchor = anchor
    p = tf.paragraphs[0]
    p.alignment = align
    if line_spacing is not None:
        p.line_spacing = line_spacing
    run = p.add_run()
    run.text = text.upper() if caps else text
    run.font.size = Pt(size)
    run.font.bold = bold
    run.font.name = font or FONT_BODY
    run.font.color.rgb = color
    if spacing is not None:
        # Letter-spacing isn't exposed by python-pptx; set it on the run XML.
        run.font._rPr.set("spc", str(int(spacing * 100)))
    return box


def add_logo(slide, prs, logo_path, width=LOGO_WIDTH, top=LOGO_TOP):
    left = prs.slide_width - width - MARGIN_X
    return slide.shapes.add_picture(str(logo_path), left, top, width=width)


def add_picture_fit(slide, prs, image_path, top=CONTENT_TOP,
                    bottom=CONTENT_BOTTOM, side_margin=MARGIN_X, align="center"):
    """Scales the image to fit the content area, preserving aspect ratio."""
    available_width = prs.slide_width - 2 * side_margin
    available_height = bottom - top

    pic = slide.shapes.add_picture(str(image_path), 0, 0)
    scale = min(available_width / pic.width, available_height / pic.height)
    pic.width = int(pic.width * scale)
    pic.height = int(pic.height * scale)
    pic.left = side_margin if align == "left" else side_margin + int((available_width - pic.width) / 2)
    pic.top = top + int((available_height - pic.height) / 2)
    return pic


def add_picture_grid(slide, prs, image_paths, ncols=2, top=CONTENT_TOP,
                     bottom=CONTENT_BOTTOM, side_margin=MARGIN_X, gutter=Inches(0.22)):
    """Each image as its own picture shape in a grid, aspect-fit inside its
    cell so nothing stretches. A short final row is centered."""
    n = len(image_paths)
    if n == 0:
        return
    nrows = (n + ncols - 1) // ncols
    available_width = prs.slide_width - 2 * side_margin
    available_height = bottom - top
    cell_w = (available_width - gutter * (ncols - 1)) / ncols
    cell_h = (available_height - gutter * (nrows - 1)) / nrows

    for row in range(nrows):
        row_images = image_paths[row * ncols:(row + 1) * ncols]
        row_width = len(row_images) * cell_w + gutter * (len(row_images) - 1)
        row_left = side_margin + (available_width - row_width) / 2
        for col, img_path in enumerate(row_images):
            pic = slide.shapes.add_picture(str(img_path), 0, 0)
            scale = min(cell_w / pic.width, cell_h / pic.height)
            pic.width = int(pic.width * scale)
            pic.height = int(pic.height * scale)
            cell_left = row_left + col * (cell_w + gutter)
            cell_top = top + row * (cell_h + gutter)
            pic.left = int(cell_left + (cell_w - pic.width) / 2)
            pic.top = int(cell_top + (cell_h - pic.height) / 2)


# --- Table borders (python-pptx has no border API, so write the XML) ------- #

_EDGE_TAGS = [("left", "a:lnL"), ("right", "a:lnR"), ("top", "a:lnT"), ("bottom", "a:lnB")]


def set_cell_borders(cell, borders):
    """borders: {"top": (hex_no_hash, width_pt), ...}. Unlisted edges are
    removed, which is how the default banded-table look gets cleared."""
    tcPr = cell._tc.get_or_add_tcPr()
    for _, tag in _EDGE_TAGS:
        for el in tcPr.findall(qn(tag)):
            tcPr.remove(el)
    index = 0
    for edge, tag in _EDGE_TAGS:
        if edge not in borders:
            continue
        color, width_pt = borders[edge]
        ln = parse_xml(
            f'<{tag} {nsdecls("a")} w="{int(width_pt * 12700)}" cap="flat" '
            f'cmpd="sng" algn="ctr">'
            f'<a:solidFill><a:srgbClr val="{color}"/></a:solidFill>'
            f'<a:prstDash val="solid"/></{tag}>'
        )
        tcPr.insert(index, ln)
        index += 1


def style_cell(cell, text, size=9.5, color=INK, bold=False, fill=None,
               align=PP_ALIGN.LEFT, borders=None, font=None):
    cell.text_frame.text = ""
    cell.margin_left = Pt(8)
    cell.margin_right = Pt(8)
    cell.margin_top = Pt(4)
    cell.margin_bottom = Pt(4)
    cell.vertical_anchor = MSO_ANCHOR.MIDDLE
    p = cell.text_frame.paragraphs[0]
    p.alignment = align
    run = p.add_run()
    run.text = text
    run.font.size = Pt(size)
    run.font.bold = bold
    run.font.name = font or FONT_BODY
    run.font.color.rgb = color
    if fill is None:
        cell.fill.background()
    else:
        cell.fill.solid()
        cell.fill.fore_color.rgb = fill
    set_cell_borders(cell, borders or {})


def plain_table(slide, prs, n_rows, n_cols, top, bottom, side_margin=MARGIN_X):
    """A table with PowerPoint's default banding/emphasis switched off, so
    the styling below is the only styling."""
    width = prs.slide_width - 2 * side_margin
    height = bottom - top
    frame = slide.shapes.add_table(n_rows, n_cols, side_margin, top, width, height)
    table = frame.table
    table.first_row = False
    table.horz_banding = False
    table.first_col = False
    table.vert_banding = False
    return frame, table, width, height

class Deck:
    """Holds the presentation plus the running footer/page state, so every
    slide is built through the same frame."""

    def __init__(self, period_label, logo_path=None):
        self.prs = Presentation()
        self.prs.slide_width = SLIDE_WIDTH
        self.prs.slide_height = SLIDE_HEIGHT
        self.blank = self.prs.slide_layouts[6]
        self.period_label = period_label
        self.logo_path = logo_path
        self.page = 0

    # -- ground ------------------------------------------------------------ #
    def _blank_slide(self, ground=GROUND):
        slide = self.prs.slides.add_slide(self.blank)
        add_rect(slide, 0, 0, self.prs.slide_width, self.prs.slide_height, ground)
        return slide

    @property
    def content_width(self):
        return self.prs.slide_width - 2 * MARGIN_X

    # -- cover ------------------------------------------------------------- #
    def cover(self, title, subtitle):
        slide = self._blank_slide()
        add_rect(slide, MARGIN_X, Inches(1.05), self.content_width, HEAD_RULE_W, NAVY)
        add_text(slide, MARGIN_X, Inches(1.28), Inches(7), Inches(0.3),
                 "Weekly operations report", PT_KICKER, NAVY_SOFT,
                 font=FONT_BODY, caps=True, spacing=180)
        add_text(slide, MARGIN_X, Inches(3.05), Inches(10.4), Inches(2.0),
                 title, PT_COVER_TITLE, NAVY, font=FONT_HEAD, bold=True,
                 line_spacing=1.02)
        add_text(slide, MARGIN_X, Inches(4.75), Inches(9), Inches(0.5),
                 subtitle, PT_BODY, INK)
        add_rect(slide, MARGIN_X, Inches(6.36), self.content_width, HAIRLINE_W, RULE)
        add_text(slide, MARGIN_X, Inches(6.52), Inches(6), Inches(0.4),
                 "Service Desk — Holland America Line", PT_FOOTER + 1, MUTED, caps=True, spacing=100)
        if self.logo_path:
            add_logo(slide, self.prs, self.logo_path, width=Inches(2.1), top=Inches(0.42))
        return slide

    # -- section divider --------------------------------------------------- #
    def divider(self, number, title, note=None):
        slide = self._blank_slide(ground=NAVY)
        add_text(slide, MARGIN_X, Inches(2.55), Inches(4), Inches(0.4),
                 f"Section {number:02d}", PT_KICKER + 1, RGBColor(0x9A, 0xA9, 0xC8),
                 caps=True, spacing=200)
        add_rect(slide, MARGIN_X, Inches(3.0), Inches(1.1), HEAD_RULE_W, GOLD)
        add_text(slide, MARGIN_X, Inches(3.35), Inches(10), Inches(1.2),
                 title, PT_DIVIDER_TITLE, WHITE, font=FONT_HEAD, bold=True)
        if note:
            add_text(slide, MARGIN_X, Inches(4.5), Inches(8.4), Inches(0.8),
                     note, PT_BODY - 3, RGBColor(0xC3, 0xCB, 0xDD))
        return slide

    # -- standard content slide -------------------------------------------- #
    def content(self, kicker, title):
        slide = self._blank_slide()
        self.page += 1
        add_text(slide, MARGIN_X, KICKER_TOP, Inches(6), Inches(0.28),
                 kicker, PT_KICKER, GOLD, caps=True, spacing=200)
        title_w = self.content_width - (LOGO_WIDTH + Inches(0.4) if self.logo_path else 0)
        add_text(slide, MARGIN_X, TITLE_TOP, title_w, Inches(0.5),
                 title, PT_TITLE, NAVY, font=FONT_HEAD, bold=True)
        add_rect(slide, MARGIN_X, HEAD_RULE_Y, self.content_width, HEAD_RULE_W, NAVY)
        add_rect(slide, MARGIN_X, FOOT_RULE_Y, self.content_width, HAIRLINE_W, RULE)
        add_text(slide, MARGIN_X, FOOTER_TOP, Inches(7), Inches(0.3),
                 f"HAL weekly report  ·  {self.period_label}", PT_FOOTER, MUTED)
        add_text(slide, self.prs.slide_width - MARGIN_X - Inches(1.2), FOOTER_TOP,
                 Inches(1.2), Inches(0.3), f"{self.page:02d}", PT_FOOTER, MUTED,
                 align=PP_ALIGN.RIGHT)
        if self.logo_path:
            add_logo(slide, self.prs, self.logo_path)
        return slide

    # -- KPI band ---------------------------------------------------------- #
    def kpis(self, kicker, title, items):
        """items: list of (label, value, note) — laid out as equal columns
        separated by hairline verticals."""
        slide = self.content(kicker, title)
        n = len(items)
        top = Inches(2.25)
        height = Inches(2.1)
        col_w = int(self.content_width / n)
        for i, (label, value, note) in enumerate(items):
            left = MARGIN_X + i * col_w
            if i:
                add_rect(slide, left - Inches(0.01), top, HAIRLINE_W, height, RULE)
            pad = Inches(0.0) if i == 0 else Inches(0.34)
            x = left + pad
            w = col_w - pad - Inches(0.3)
            add_text(slide, x, top, w, Inches(0.3), label, PT_KPI_LABEL, MUTED,
                     caps=True, spacing=120)
            add_text(slide, x, top + Inches(0.44), w, Inches(1.0), value,
                     PT_KPI_VALUE, NAVY, font=FONT_HEAD, bold=True)
            add_text(slide, x, top + Inches(1.52), w, Inches(0.6), note,
                     PT_KPI_LABEL + 1, INK, line_spacing=1.25)
        return slide

    def save(self, path):
        self.prs.save(path)

def add_pivot_table(deck, slide, pivot_df):
    """Primary Configuration Item / Summary / Number, with repeated items
    merged per group, a navy header and hairline row rules."""
    prs = deck.prs
    n_rows = len(pivot_df) + 1
    frame, table, width, height = plain_table(slide, prs, n_rows, 3, CONTENT_TOP, CONTENT_BOTTOM)

    table.columns[0].width = Inches(3.3)
    table.columns[2].width = Inches(1.1)
    table.columns[1].width = Emu(int(width - Inches(3.3) - Inches(1.1)))

    header_h = Inches(0.42)
    table.rows[0].height = header_h
    if n_rows > 1:
        body_h = int((height - header_h) / (n_rows - 1))
        for row in list(table.rows)[1:]:
            row.height = body_h

    headers = ["Primary configuration item", "Summary", "Tickets"]
    for col, header in enumerate(headers):
        style_cell(table.cell(0, col), header, size=10.5, bold=True, color=WHITE,
                   fill=NAVY, font=FONT_HEAD,
                   align=PP_ALIGN.RIGHT if col == 2 else PP_ALIGN.LEFT)

    group_start_row = 1
    last_item = None
    group_index = 0
    for i, row_data in pivot_df.reset_index(drop=True).iterrows():
        r = i + 1
        item = str(row_data["Primary Configuration Item"])
        new_group = item != last_item
        if new_group:
            if last_item is not None and group_start_row < r - 1:
                table.cell(group_start_row, 0).merge(table.cell(r - 1, 0))
            group_start_row = r
            last_item = item
            group_index += 1
        tint = BAND if group_index % 2 == 0 else None
        style_cell(table.cell(r, 0), item if new_group else "", size=9.5,
                   bold=new_group, color=NAVY if new_group else INK, fill=tint,
                   font=FONT_HEAD if new_group else FONT_BODY,
                   borders={"top": (RULE_HEX, 0.75)})
        style_cell(table.cell(r, 1), str(row_data["Summary"]), size=9.5, fill=tint,
                   borders={"top": (RULE_HEX, 0.75)})
        style_cell(table.cell(r, 2), str(row_data["Number"]), size=9.5, bold=True,
                   color=NAVY, fill=tint, align=PP_ALIGN.RIGHT,
                   borders={"top": (RULE_HEX, 0.75)})
    if last_item is not None and group_start_row < n_rows - 1:
        table.cell(group_start_row, 0).merge(table.cell(n_rows - 1, 0))
    return frame


def add_records_table(deck, slide, df):
    """Flat table of every column in `df`, one row per record."""
    prs = deck.prs
    n_rows = len(df) + 1
    n_cols = len(df.columns)
    frame, table, width, height = plain_table(slide, prs, n_rows, n_cols,
                                              CONTENT_TOP, CONTENT_BOTTOM)

    col_width = Emu(int(width / n_cols))
    for col in table.columns:
        col.width = col_width

    header_h = Inches(0.4)
    table.rows[0].height = header_h
    if n_rows > 1:
        body_h = int((height - header_h) / (n_rows - 1))
        for row in list(table.rows)[1:]:
            row.height = body_h

    for col, header in enumerate(df.columns):
        style_cell(table.cell(0, col), str(header), size=9.5, bold=True,
                   color=WHITE, fill=NAVY, font=FONT_HEAD)

    for i, row_data in df.reset_index(drop=True).iterrows():
        r = i + 1
        tint = BAND if r % 2 == 0 else None
        for col, value in enumerate(row_data):
            style_cell(table.cell(r, col), "" if pd.isna(value) else str(value),
                       size=8.5, fill=tint, borders={"top": (RULE_HEX, 0.75)})
    return frame


def add_statement(deck, slide, headline, support=None):
    """A single sentence given the whole slide — used for the No CI Found
    streak, which is a one-line story, not a bullet."""
    add_rect(slide, MARGIN_X, Inches(2.5), Inches(1.1), HEAD_RULE_W, GOLD)
    add_text(slide, MARGIN_X, Inches(2.9), Inches(9.8), Inches(2.0),
             headline, 34, NAVY, font=FONT_HEAD, bold=True, line_spacing=1.15)
    if support:
        add_text(slide, MARGIN_X, Inches(4.9), Inches(8.6), Inches(0.8),
                 support, PT_BODY - 3, INK)

def build_presentation(cover_title, period_label, kpi_items, weekday_chart,
                       hourly_count_charts, hourly_wait_charts, apps_chart,
                       pivot_df, noci_text, noci_df, csat_charts, csat_summary,
                       output_path, logo_path=None):
    deck = Deck(period_label, logo_path=logo_path)

    # 1 — Cover
    deck.cover(cover_title, f"Reporting period {period_label}")

    # 2 — At a glance
    deck.kpis("Summary", "The week at a glance", kpi_items)

    # 3 — Call volume
    if SECTION_DIVIDERS:
        deck.divider(1, "Call volume", "Calls received across the HAL queues, by day and by hour.")
    slide = deck.content("Call volume", "Calls received by weekday")
    add_picture_fit(slide, deck.prs, weekday_chart)

    for i, group in enumerate((hourly_count_charts[:4], hourly_count_charts[4:])):
        slide = deck.content("Call volume", f"Calls offered by hour  ({i + 1} of 2)")
        add_picture_grid(slide, deck.prs, group)

    # 4 — Waiting time
    if SECTION_DIVIDERS:
        deck.divider(2, "Waiting time", "Average time in queue before a call was answered.")
    for i, group in enumerate((hourly_wait_charts[:4], hourly_wait_charts[4:])):
        slide = deck.content("Waiting time", f"Average waiting time by hour  ({i + 1} of 2)")
        add_picture_grid(slide, deck.prs, group)

    # 5 — Applications
    if SECTION_DIVIDERS:
        deck.divider(3, "Applications", "Which configuration items generated the week's tickets.")
    slide = deck.content("Applications", f"Top {TOP_N_APPS} primary configuration items")
    add_picture_fit(slide, deck.prs, apps_chart)

    slide = deck.content("Applications", "Ticket detail by configuration item")
    add_pivot_table(deck, slide, pivot_df)

    slide = deck.content("Applications", "No CI found")
    if noci_text is not None:
        add_statement(deck, slide, noci_text,
                      "Every ticket raised this week was mapped to a configuration item.")
    else:
        add_records_table(deck, slide, noci_df)

    # 6 — Satisfaction
    if SECTION_DIVIDERS:
        deck.divider(4, "Customer satisfaction", "Survey responses for the reporting period.")
    slide = deck.content("Customer satisfaction", "C-SAT results")
    if csat_summary:
        add_text(slide, MARGIN_X, Inches(1.78), Inches(11), Inches(0.4),
                 csat_summary, PT_BODY - 3, INK)
        add_picture_grid(slide, deck.prs, list(csat_charts), ncols=2, top=Inches(2.42))
    else:
        add_picture_grid(slide, deck.prs, list(csat_charts), ncols=2)

    deck.save(output_path)

def main():
    parser = argparse.ArgumentParser(description="Generate the HAL weekly PPTX report.")
    parser.add_argument("--csv", default=str(DEFAULT_CSV_PATH),
                        help="Path to the calls CSV export, or a folder containing it "
                             f"(default: {DEFAULT_CSV_PATH}).")
    parser.add_argument("--excel", default=str(DEFAULT_EXCEL_PATH),
                        help=f"Path to the tickets Excel export (default: {DEFAULT_EXCEL_PATH}).")
    parser.add_argument("--output", default=None,
                        help="Output .pptx path (default: "
                             f"'{DEFAULT_OUTPUT_PATH / 'HAL Weekly Report - <Start Date> to <End Date>.pptx'}', "
                             "built from the dates you enter).")
    parser.add_argument("--logo", default=str(DEFAULT_LOGO_PATH),
                        help=f"Logo image placed in the top-right corner of every slide "
                             f"(default: {DEFAULT_LOGO_PATH}). Skipped if not found.")
    parser.add_argument("--noci", default=str(DEFAULT_NOCI_PATH),
                        help="Path to the No CI Found Excel export, used only when you answer "
                             f"'no' to the No CI Found prompt (default: {DEFAULT_NOCI_PATH}).")
    args = parser.parse_args()

    excel_path = Path(args.excel)
    logo_path = Path(args.logo)
    if not logo_path.is_file():
        print(f"Logo not found at {logo_path}, skipping it.")
        logo_path = None
    csv_path = resolve_csv_path(args.csv)
    if not csv_path.exists():
        raise FileNotFoundError(f"CSV file not found: {csv_path}")
    if not excel_path.exists():
        raise FileNotFoundError(f"Excel file not found: {excel_path}")
    if excel_path.is_dir():
        raise IsADirectoryError(
            f"--excel points at a folder, not a file: {excel_path}\n"
            "Add the actual Excel filename to the end of the path."
        )
    csat_path = resolve_csat_path(csv_path.parent)
    if csat_path is None:
        raise FileNotFoundError(
            f"No C-SAT file (an .xlsx with 'C-SAT' in its name) found in {csv_path.parent}"
        )

    start_date, end_date, report_title = get_date_inputs()

    no_ci_found_this_week = prompt_yes_no("No CI Found This Week")
    if no_ci_found_this_week:
        streak = load_streak() + 1
        save_streak(streak)
        noci_text = f'For the {ordinal(streak)} straight week, there were no "No CI Found" tickets.'
        noci_df = None
    else:
        save_streak(0)
        noci_path = Path(args.noci)
        if not noci_path.exists():
            raise FileNotFoundError(f"No CI Found file not found: {noci_path}")
        noci_df = load_noci_records(noci_path)
        noci_text = None

    if args.output is not None:
        output_path = Path(args.output)
        if output_path.is_dir() or not output_path.suffix:
            raise IsADirectoryError(
                f"--output points at a folder, not a file: {output_path}\n"
                "Add a .pptx filename to the end of the path."
            )
    else:
        output_path = DEFAULT_OUTPUT_PATH / f"HAL Weekly Report - {format_date(start_date)} to {format_date(end_date)}.pptx"

    calls = load_calls(csv_path, start_date, end_date)
    apps = load_apps(excel_path)
    pivot_df = load_pivot_table(excel_path)
    ticket_total = load_ticket_total(excel_path)
    hours = _hour_range(calls)

    period_label = format_date_range(start_date, end_date)
    cover_title = (
        f"HAL weekly report\n{start_date.day} {calendar.month_abbr[start_date.month]} – "
        f"{end_date.day} {calendar.month_abbr[end_date.month]} {end_date.year}"
    )

    with tempfile.TemporaryDirectory() as tmp:
        tmp = Path(tmp)

        weekday_chart = tmp / "weekday_counts.png"
        make_weekday_count_chart(calls, weekday_chart)

        hourly_count_charts = []
        hourly_wait_charts = []
        for day in WEEKDAYS:
            chart_date = date_for_weekday(start_date, end_date, day)
            count_path = tmp / f"hourly_count_{day}.png"
            make_single_hour_chart(calls, day, hours, count_path, mode="count", chart_date=chart_date)
            hourly_count_charts.append(count_path)
            wait_path = tmp / f"hourly_wait_{day}.png"
            make_single_hour_chart(calls, day, hours, wait_path, mode="avg_wait", chart_date=chart_date)
            hourly_wait_charts.append(wait_path)

        apps_chart = tmp / "apps.png"
        make_apps_chart(apps, apps_chart)

        votes_cats, votes_series = extract_named_chart_data(csat_path, CSAT_SHEET_NAME, CSAT_VOTES_CHART)
        csat_votes_chart = tmp / "csat_votes.png"
        make_csat_votes_chart(votes_cats, votes_series, csat_votes_chart)

        overall_cats, overall_series = extract_named_chart_data(csat_path, CSAT_SHEET_NAME, CSAT_OVERALL_CHART)
        csat_overall_chart = tmp / "csat_overall.png"
        make_csat_overall_chart(overall_cats, overall_series, csat_overall_chart)

        # --- headline numbers for the "at a glance" slide ------------------ #
        counts_by_day = calls["Weekday"].value_counts().reindex(WEEKDAYS, fill_value=0)
        busiest_day = counts_by_day.idxmax() if counts_by_day.sum() else "—"
        avg_wait = calls["TotalQueueSeconds"].mean()
        try:
            csat_value = float(overall_series[0][1][0] or 0)
        except (IndexError, TypeError, ValueError):
            csat_value = 0.0
        top_app = apps.index[0] if len(apps) else "—"
        top_app_n = int(apps.iloc[0]) if len(apps) else 0

        kpi_items = [
            (f"Calls received  ·  {period_label}", f"{len(calls):,}",
             f"Busiest day was {busiest_day}, with {int(counts_by_day.max()) if counts_by_day.sum() else 0} calls."),
            ("Average time in queue", seconds_to_mmss(avg_wait),
             "Mean wait before a call was answered."),
            ("Tickets logged", f"{ticket_total:,}",
             f"Largest single item: {top_app} ({top_app_n})."),
            ("C-SAT rating", f"{csat_value:.2f}",
             f"Out of 5, from this week's survey responses."),
        ]
        csat_summary = (
            f"Overall satisfaction of {csat_value:.2f} out of 5 for {period_label}."
        )

        build_presentation(
            cover_title,
            period_label,
            kpi_items,
            weekday_chart,
            hourly_count_charts,
            hourly_wait_charts,
            apps_chart,
            pivot_df,
            noci_text,
            noci_df,
            [csat_overall_chart, csat_votes_chart],
            csat_summary,
            output_path,
            logo_path=logo_path,
        )

    print(f"Report saved to {output_path}")


if __name__ == "__main__":
    main()
